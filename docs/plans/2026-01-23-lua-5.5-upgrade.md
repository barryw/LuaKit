//

//  LuaKit
//
//  Created by Claude on 2026-01-23.
//

import Foundation

/// Libraries that can be selectively loaded into a Lua state.
///
/// Use this with `LuaState(libraries:)` to create sandboxed Lua environments
/// that don't have access to potentially dangerous functionality like file I/O
/// or OS commands.
///
/// Example:
/// ```swift
/// // Full access (default)
/// let lua = try LuaState()
///
/// // Sandboxed - no io, os, debug, package
/// let sandbox = try LuaState(libraries: .sandboxed)
///
/// // Custom selection
/// let custom = try LuaState(libraries: [.base, .string, .math])
/// ```
public struct LuaLibrary: OptionSet, Sendable {
    public let rawValue: Int

    public init(rawValue: Int) {
        self.rawValue = rawValue
    }

    /// Basic functions: print, tonumber, tostring, type, etc.
    public static let base = LuaLibrary(rawValue: 1 << 0)

    /// Coroutine manipulation functions
    public static let coroutine = LuaLibrary(rawValue: 1 << 1)

    /// Table manipulation functions
    public static let table = LuaLibrary(rawValue: 1 << 2)

    /// File I/O functions (potentially dangerous)
    public static let io = LuaLibrary(rawValue: 1 << 3)

    /// Operating system facilities (potentially dangerous)
    public static let os = LuaLibrary(rawValue: 1 << 4)

    /// String manipulation functions
    public static let string = LuaLibrary(rawValue: 1 << 5)

    /// Mathematical functions
    public static let math = LuaLibrary(rawValue: 1 << 6)

    /// UTF-8 support functions
    public static let utf8 = LuaLibrary(rawValue: 1 << 7)

    /// Debug facilities (potentially dangerous)
    public static let debug = LuaLibrary(rawValue: 1 << 8)

    /// Package/module loading (potentially dangerous)
    public static let package = LuaLibrary(rawValue: 1 << 9)

    /// All standard libraries
    public static let all: LuaLibrary = [
        .base, .coroutine, .table, .io, .os,
        .string, .math, .utf8, .debug, .package
    ]

    /// Safe subset for sandboxed execution.
    /// Excludes: io, os, debug, package
    public static let sandboxed: LuaLibrary = [
        .base, .coroutine, .table, .string, .math, .utf8
    ]
}
```

/// Creates a new Lua state with only the specified libraries loaded.
///
/// Use this to create sandboxed Lua environments that don't have access
/// to potentially dangerous functionality.
///
/// - Parameter libraries: The set of libraries to load. Defaults to `.all`.
/// - Throws: `LuaError.memoryAllocation` if state creation fails.
///
/// Example:
/// ```swift
/// // Sandboxed state without io, os, debug, package
/// let sandbox = try LuaState(libraries: .sandboxed)
///
/// // Custom selection
/// let custom = try LuaState(libraries: [.base, .string, .math])
/// ```
public init(libraries: LuaLibrary) throws {
    guard let state = luaL_newstate() else {
        throw LuaError.memoryAllocation
    }
    self.L = state

    // Open selected libraries using Lua 5.5's luaL_openselectedlibs
    // or fall back to individual library openers
    openSelectedLibraries(libraries)

    setupPrintCapture()
    registerArrayProxyTypes()
}

private func openSelectedLibraries(_ libraries: LuaLibrary) {
    // Lua 5.5 provides luaL_openselectedlibs, but we need to map our flags
    // to Lua's library constants. For compatibility, we open libraries individually.

    if libraries.contains(.base) {
        luaL_requiref(L, LUA_GNAME, luaopen_base, 1)
        lua_pop(L, 1)
    }
    if libraries.contains(.coroutine) {
        luaL_requiref(L, LUA_COLIBNAME, luaopen_coroutine, 1)
        lua_pop(L, 1)
    }
    if libraries.contains(.table) {
        luaL_requiref(L, LUA_TABLIBNAME, luaopen_table, 1)
        lua_pop(L, 1)
    }
    if libraries.contains(.io) {
        luaL_requiref(L, LUA_IOLIBNAME, luaopen_io, 1)
        lua_pop(L, 1)
    }
    if libraries.contains(.os) {
        luaL_requiref(L, LUA_OSLIBNAME, luaopen_os, 1)
        lua_pop(L, 1)
    }
    if libraries.contains(.string) {
        luaL_requiref(L, LUA_STRLIBNAME, luaopen_string, 1)
        lua_pop(L, 1)
    }
    if libraries.contains(.math) {
        luaL_requiref(L, LUA_MATHLIBNAME, luaopen_math, 1)
        lua_pop(L, 1)
    }
    if libraries.contains(.utf8) {
        luaL_requiref(L, LUA_UTF8LIBNAME, luaopen_utf8, 1)
        lua_pop(L, 1)
    }
    if libraries.contains(.debug) {
        luaL_requiref(L, LUA_DBLIBNAME, luaopen_debug, 1)
        lua_pop(L, 1)
    }
    if libraries.contains(.package) {
        luaL_requiref(L, LUA_LOADLIBNAME, luaopen_package, 1)
        lua_pop(L, 1)
    }
}
```

import XCTest
@testable import LuaKit

/// Smoke tests verifying Lua 5.5 features work correctly
final class Lua55FeatureTests: XCTestCase {

    func testTableCreate() throws {
        let lua = try LuaState()
        // table.create is new in Lua 5.5
        let result = try lua.execute("""
            local t = table.create(10)
            return type(t)
        """)
        XCTAssertEqual(result.first, .string("table"))
    }

    func testTableCreateWithInitialValue() throws {
        let lua = try LuaState()
        let result = try lua.execute("""
            local t = table.create(5, 0)
            return t[1], t[5]
        """)
        XCTAssertEqual(result.count, 2)
        XCTAssertEqual(result[0], .integer(0))
        XCTAssertEqual(result[1], .integer(0))
    }

    func testLargeArraysWork() throws {
        let lua = try LuaState()
        // Lua 5.5 has ~60% smaller arrays
        let result = try lua.execute("""
            local t = {}
            for i = 1, 10000 do
                t[i] = i
            end
            return #t
        """)
        XCTAssertEqual(result.first, .integer(10000))
    }

    func testLuaVersionIs55() throws {
        let lua = try LuaState()
        let result = try lua.execute("return _VERSION")
        XCTAssertEqual(result.first, .string("Lua 5.5"))
    }
}
